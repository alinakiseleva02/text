Запрос 5: Получить список всех пользователей, которые оценили больше n работ

   CREATE OR REPLACE FUNCTION users_with_more_than_n_ratings(n INT)
     RETURNS TABLE (user_id INT)
     AS $$
     BEGIN
       RETURN QUERY
       SELECT user_id
       FROM ratings
       GROUP BY user_id
       HAVING COUNT(*) > n;
     END;
     $$ LANGUAGE plpgsql;
